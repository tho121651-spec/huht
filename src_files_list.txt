// Component bảo vệ Route
function ProtectedRoute({ children }) {
  const { user } = useAuth()
  const storedUser = user || JSON.parse(localStorage.getItem('user'))

  return storedUser ? children : <Navigate to="/Account/Login" replace />
}

function AppRoutes() {
  return (
    <Routes>
      {/* Route Đăng nhập & Đăng xuất */}

      <Route path="/Account/Logout" element={<Logout />} />

      {/* Các Route cần đăng nhập */}
      <Route
        element={
          <ProtectedRoute>
            <MainLayout />
          </ProtectedRoute>
        }
      >
        <Route path="/" element={<Navigate to="/News" replace />} />

import { useEffect, useState } from 'react'

const WEEKDAYS = ['Chủ nhật', 'Thứ 2', 'Thứ 3', 'Thứ 4', 'Thứ 5', 'Thứ 6', 'Thứ 7']

function formatVietnameseDate(date) {
  return `${WEEKDAYS[date.getDay()]}, ngày ${date.getDate()} tháng ${date.getMonth() + 1} năm ${date.getFullYear()}`
}

export default function Header() {
  const [today, setToday] = useState(() => formatVietnameseDate(new Date()))

  useEffect(() => {
    const timer = setInterval(() => setToday(formatVietnameseDate(new Date())), 60000)
    return () => clearInterval(timer)
  }, [])

  return (
    <div className="row hitec-title">
      <div className="col-xs-8 nopadding">
        <h1>TRƯỜNG DU LỊCH - ĐẠI HỌC HUẾ</h1>
      </div>
      <div className="col-xs-4 text-right" style={{ color: '#cecece', lineHeight: '28px' }}>
        {today}
      </div>
    </div>
  )
}

import { useEffect, useRef, useState } from 'react'
import { Link, useNavigate } from 'react-router-dom'
import { mainMenu } from '../data/menu.js'

export default function Navbar() {
  const [openIndex, setOpenIndex] = useState(null)
  const [collapsed, setCollapsed] = useState(true)
  const navRef = useRef(null)
  const navigate = useNavigate()

  useEffect(() => {
    function handleOutside(event) {
      if (navRef.current && !navRef.current.contains(event.target)) setOpenIndex(null)
    }
    function handleEsc(event) {
      if (event.key === 'Escape') setOpenIndex(null)
    }
    document.addEventListener('mousedown', handleOutside)
    document.addEventListener('keydown', handleEsc)
    return () => {
      document.removeEventListener('mousedown', handleOutside)
      document.removeEventListener('keydown', handleEsc)
    }
  }, [])

  function go(to) {
    setOpenIndex(null)
    setCollapsed(true)
    navigate(to)
  }

  return (
    <div className="row">
      <div className="col-xs-12 nopadding">
        <nav className="navbar navbar-default hitec-navbar" ref={navRef}>
          <div className="navbar-header">
            <button
              type="button"
              className="pull-left navbar-toggle"
              onClick={() => setCollapsed((value) => !value)}
            >
              <span className="sr-only">Toggle navigation</span>
              <span className="icon-bar" />
              <span className="icon-bar" />
              <span className="icon-bar" />
            </button>
            <Link className="navbar-brand hitec-logo" to="/News">
              <img src={`${import.meta.env.BASE_URL}images/logo.png`} alt="logo" />
            </Link>
          </div>

          {/* menu main */}
          <div className={`nav collapse navbar-collapse ${collapsed ? '' : 'in'}`} id="menuMain">
            <ul className="nav navbar-nav">
              {mainMenu.map((menu, index) => (
                <li key={menu.label} className={`dropdown ${openIndex === index ? 'open' : ''}`}>
                  <a
                    href="#"
                    className="dropdown-toggle"
                    aria-expanded={openIndex === index}
                    onClick={(event) => {
                      event.preventDefault()
                      setOpenIndex(openIndex === index ? null : index)
                    }}
                  >
                    {menu.label} <b className="caret" />
                  </a>
                  <ul className="dropdown-menu">
                    {menu.items.map((item, i) => {
                      if (item.type === 'header') {
                        return <li key={`h-${i}`} className="dropdown-header">{item.label}</li>
                      }
                      if (item.type === 'divider') {
                        return <li key={`d-${i}`} className="divider" />
                      }
                      return (
                        <li key={item.to}>
                          <a
                            href={`${import.meta.env.BASE_URL.replace(/\/$/, '')}${item.to}`}
                            onClick={(event) => {
                              event.preventDefault()
                              go(item.to)
                            }}
                          >
                            {item.label}
                          </a>
                        </li>
                      )
                    })}
                  </ul>
                </li>
              ))}
            </ul>
          </div>
        </nav>
      </div>
    </div>
  )
}

import React, { createContext, useContext, useState } from 'react'
import { mockUsers } from '../data/menu.js' // Đường dẫn tới file menu/mockUsers của bạn

const AuthContext = createContext()

export const AuthProvider = ({ children }) => {
  const [user, setUser] = useState(() => {
    const saved = localStorage.getItem('user')
    return saved ? JSON.parse(saved) : null
  })

  const login = (studentId, password) => {
    // So sánh ép kiểu String và loại bỏ khoảng trắng thừa
    const foundUser = mockUsers.find(
      (u) =>
        String(u.studentId).trim() === String(studentId).trim() &&
        String(u.password).trim() === String(password).trim()
    )

    if (foundUser) {
      setUser(foundUser)
      localStorage.setItem('user', JSON.stringify(foundUser))
      return { success: true }
    } else {
      return { success: false, message: 'Mã sinh viên hoặc mật khẩu không chính xác!' }
    }
  }

  const logout = () => {
    setUser(null)
    localStorage.removeItem('user')
  }

  return (
    <AuthContext.Provider value={{ user, login, logout }}>
      {children}
    </AuthContext.Provider>
  )
}

export const useAuth = () => useContext(AuthContext)

export default function Login() {
  const [formData, setFormData] = useState({ studentId: '', password: '' })
  const [error, setError] = useState('')
  const { login } = useAuth()
  const navigate = useNavigate()

  const baseUrl = import.meta.env.BASE_URL

  const handleChange = (e) => {
    const { name, value } = e.target
    setFormData((prev) => ({ ...prev, [name]: value }))
  }

  const handleSubmit = (e) => {
    e.preventDefault()
    setError('')

    // 1. Kiểm tra rỗng
    if (!formData.studentId.trim() || !formData.password.trim()) {
      setError('Vui lòng nhập đầy đủ Mã sinh viên và Mật khẩu!')

    if (res.success) {
      // 3. Chuyển hướng tới trang Tin tức (/News)
      navigate('/News', { replace: true })
    } else {
      setError(res.message)
    }
  }

  return (
    <div className="login-body">
      <div className="hitec-signin">

        <div className="hitec-col-image text-center">

import { useEffect } from 'react'
import { useNavigate } from 'react-router-dom'
import { useAuth } from '../context/AuthContext'

export default function Logout() {
  const { logout } = useAuth()
  const navigate = useNavigate()

  useEffect(() => {
    // Hỏi xác nhận đúng như giao diện gốc
    const isConfirmed = window.confirm('Thoát khỏi hệ thống')

    if (isConfirmed) {
      logout() // Xóa state user + localStorage
      navigate('/Account/Login', { replace: true })
    } else {
      // Nếu nhấn Cancel thì quay lại trang Tin tức
      navigate('/News', { replace: true })
    }
  }, [logout, navigate])

  return null
}
